\documentclass[border=12pt]{standalone}

\usepackage{tikz}
\usetikzlibrary{arrows.meta, positioning, shapes.geometric, calc}

\begin{document}

\begin{tikzpicture}[
    node distance=0.55cm,
    
    activity/.style={
        rectangle,
        draw=black,
        thick,
        minimum width=3.2cm,
        minimum height=0.7cm,
        align=center,
        font=\small,
        rounded corners=8pt
    },
    
    decision/.style={
        diamond,
        draw=black,
        thick,
        minimum width=2.1cm,
        minimum height=1.0cm,
        align=center,
        font=\small,
        inner sep=1pt,
        aspect=2
    },
    
    startend/.style={
        circle,
        draw=black,
        thick,
        minimum size=0.45cm,
        fill=black
    },
    
    arr/.style={
        -{Stealth[length=2.2mm]},
        thick
    },
    
    note/.style={
        font=\footnotesize,
        text=gray,
        align=left
    }
]

% =========================================================
% Start
% =========================================================

\node[startend] (start) {};

% =========================================================
% User actions
% =========================================================

\node[activity, below=0.65cm of start] (capture)
    {Capture photos};

\node[activity, below=of capture] (level)
    {Level camera\\(sensors)};

% =========================================================
% Profile extraction
% =========================================================

\node[activity, below=0.75cm of level] (seg)
    {Segment sky (U-Net)};

\node[activity, below=of seg] (refine)
    {Refine boundary};

\node[activity, below=of refine] (extract)
    {Extract skyline profile};

% =========================================================
% Validation
% =========================================================

\node[decision, below=0.8cm of extract] (valid)
    {Valid?};

\node[note, right=0.5cm of valid]
    {Coverage $>180^\circ$\\
     Std. deviation $>1.5^\circ$};

% =========================================================
% Fusion and matching
% =========================================================

\node[activity, below=0.9cm of valid] (fuse)
    {Fuse multi-crop profiles};

\node[activity, below=0.75cm of fuse] (coarse)
    {Coarse scan};

\node[activity, below=of coarse] (refine2)
    {Local refinement};

\node[activity, below=of refine2] (dtw)
    {FastDTW re-ranking};

\node[activity, below=of dtw] (rrf)
    {RRF fusion};

% =========================================================
% Output
% =========================================================

\node[activity, below=0.75cm of rrf] (result)
    {Location estimate};

\node[startend, below=0.65cm of result] (end) {};

% =========================================================
% Main flow
% =========================================================

\draw[arr] (start) -- (capture);
\draw[arr] (capture) -- (level);

\draw[arr] (level) -- (seg);
\draw[arr] (seg) -- (refine);
\draw[arr] (refine) -- (extract);

\draw[arr] (extract) -- (valid);

% =========================================================
% Decision: YES
% =========================================================

\draw[arr]
    (valid.south)
    -- node[right, font=\footnotesize] {Yes}
    (fuse.north);

% =========================================================
% Decision: NO / RETRY
% =========================================================

\coordinate (retry) at ($(valid.west)+(-1.4,0)$);

\draw[arr]
    (valid.west)
    -- node[above, font=\footnotesize, pos=0.45] {No}
    (retry)
    |- (capture.west);

% =========================================================
% Remaining flow
% =========================================================

\draw[arr] (fuse) -- (coarse);
\draw[arr] (coarse) -- (refine2);
\draw[arr] (refine2) -- (dtw);
\draw[arr] (dtw) -- (rrf);

\draw[arr] (rrf) -- (result);
\draw[arr] (result) -- (end);

\end{tikzpicture}

\end{document}
